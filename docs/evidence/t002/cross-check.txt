$ cmp <(grep -v "^elapsed_ms=" docs/evidence/t002/run1.txt) <(grep -v "^elapsed_ms=" docs/evidence/t002/run2.txt)

$ diff <(grep -v "^elapsed_ms=" docs/evidence/t002/run1.txt) <(grep -v "^elapsed_ms=" docs/evidence/t002/run2.txt) && echo DIFF_EXIT=0 || echo DIFF_EXIT=$?
DIFF_EXIT=0（无差异）
